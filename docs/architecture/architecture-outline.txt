// src/lib/scoring.ts
export interface ScoringResult {
  score: number;              // 0.0 to 1.0
  passed: boolean;
  criteriaResults: Record<string, boolean>;
  passedCount: number;
  totalCount: number;
}

export function calculateScore(
  criteriaResults: Record<string, boolean>,
  passingThreshold: number
): ScoringResult {
  const values = Object.values(criteriaResults);
  const passedCount = values.filter(Boolean).length;
  const totalCount = values.length;
  const score = totalCount > 0 ? passedCount / totalCount : 0;
  const passed = score >= passingThreshold;
  
  return {
    score,
    passed,
    criteriaResults,
    passedCount,
    totalCount,
  };
}

PROGRESSION GATING:
export function canAccessModule(
  moduleId: string,
  modules: ModuleMetadata[],
  progress: ProgressState
): boolean {
  const module = modules.find(m => m.id === moduleId);
  if (!module) return false;
  
  // First module is always unlocked
  if (module.order === 1) return true;
  
  // Check all prerequisites completed
  for (const prereqId of module.prerequisites) {
    const prereqProgress = progress.modules[prereqId];
    if (!prereqProgress || prereqProgress.status !== 'completed') {
      return false;
    }
  }
  
  // Check previous module completed (sequential)
  const prevModule = modules.find(m => m.order === module.order - 1);
  if (prevModule) {
    const prevProgress = progress.modules[prevModule.id];
    if (!prevProgress || prevProgress.status !== 'completed') {
      return false;
    }
  }
  
  return true;
}

// src/components/ErrorBoundary.tsx
export class ErrorBoundary extends React.Component<
  { children: React.ReactNode; fallback?: React.ComponentType<FallbackProps> },
  { hasError: boolean; error: Error | null }
> {
  static getDerivedStateFromError(error: Error) {
    return { hasError: true, error };
  }
  
  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
    console.error('Error boundary caught:', error, errorInfo);
    // Could send to analytics/monitoring service
  }
  
  render() {
    if (this.state.hasError) {
      const Fallback = this.props.fallback || DefaultErrorFallback;
      return <Fallback error={this.state.error} reset={() => window.location.reload()} />;
    }
    return this.props.children;
  }
}

export function ContentView({ moduleId, contentId }: ContentViewProps) {
  const [content, setContent] = useState<ParsedContent | null>(null);
  const [error, setError] = useState<Error | null>(null);
  const [loading, setLoading] = useState(true);
  
  useEffect(() => {
    loadContent(moduleId, contentId)
      .then(setContent)
      .catch(err => {
        console.error('Failed to load content:', err);
        setError(err);
      })
      .finally(() => setLoading(false));
  }, [moduleId, contentId]);
  
  if (loading) return <LoadingSpinner />;
  if (error) return <ContentErrorFallback error={error} onRetry={() => window.location.reload()} />;
  if (!content) return <ContentNotFound />;
  
  return <MarkdownRenderer content={content.content} />;
}

function handleStorageError(error: Error): void {
  if (error.name === 'QuotaExceededError') {
    // Show dialog: "Storage full. Please export your progress."
    showNotification({
      type: 'error',
      title: 'Storage Full',
      message: 'Please export your progress to backup your data.',
      actions: [

      ]
    });
  } else if (error.name === 'SecurityError') {
    // Private browsing or storage disabled
    showNotification({
      type: 'warning',
      title: 'Storage Unavailable',
      message: 'Your browser settings prevent saving progress. Enable storage or use export/import.',
    });
  } else {
    // Generic storage error
    showNotification({
      type: 'error',
      title: 'Failed to Save Progress',
      message: 'Your progress could not be saved. Try refreshing the page.',
    });
  }
}

// src/lib/markdownParser.ts
export function parseMarkdown(raw: string): ParsedContent {
  try {
    const { data, content } = parseFrontmatter(raw);
    
    // Validate frontmatter schema
    if (!isValidFrontmatter(data)) {
      throw new Error('Invalid frontmatter schema');
    }
    
    return {
      frontmatter: data as ContentFrontmatter,
      content,
      raw,
    };
  } catch (error) {
    console.error('Markdown parsing failed:', error);
    // Return safe fallback
    return {
      frontmatter: {
        id: 'unknown',
        type: 'exercise',
        title: 'Content Error',
        duration: 0,
        skills: [],
        difficulty: 'beginner',
      },
      content: '**Error**: This content could not be loaded properly.',
      raw,
    };
  }
}

CONTENT LOADING STRATEGY:

// src/lib/contentLoader.ts
// Discover all module.json files at build time
const moduleManifests = import.meta.glob<{ default: ModuleMetadata }>(
  '/src/content/modules/*/module.json',
  { eager: false } // Lazy load
);

// Discover all markdown files at build time
const markdownFiles = import.meta.glob<string>(
  '/src/content/modules/**/*.md',
  { 
    eager: false,
    query: '?raw',
    import: 'default'
  }
);

// Load module metadata (cached in memory after first load)
const moduleCache = new Map<string, ModuleMetadata>();

export async function loadModule(moduleId: string): Promise<ModuleMetadata> {
  if (moduleCache.has(moduleId)) {
    return moduleCache.get(moduleId)!;
  }
  
  const path = `/src/content/modules/${moduleId}/module.json`;
  const loader = moduleManifests[path];
  
  if (!loader) {
    throw new Error(`Module not found: ${moduleId}`);
  }
  
  const module = await loader();
  moduleCache.set(moduleId, module.default);
  return module.default;
}

// Load markdown content (parsed and cached)
const contentCache = new Map<string, ParsedContent>();

export async function loadContent(
  moduleId: string,
  relativePath: string
): Promise<ParsedContent> {
  const fullPath = `/src/content/modules/${moduleId}/${relativePath}`;
  
  if (contentCache.has(fullPath)) {
    return contentCache.get(fullPath)!;
  }
  
  const loader = markdownFiles[fullPath];
  
  if (!loader) {
    throw new Error(`Content not found: ${fullPath}`);
  }
  
  const raw = await loader();
  const parsed = parseMarkdown(raw);
  contentCache.set(fullPath, parsed);
  return parsed;
}

// Preload next module/content for faster navigation
export function preloadModule(moduleId: string): void {
  loadModule(moduleId).catch(err => {
    console.warn('Preload failed:', moduleId, err);
  });
}

export function preloadContent(moduleId: string, relativePath: string): void {
  loadContent(moduleId, relativePath).catch(err => {
    console.warn('Preload failed:', relativePath, err);
  });
}

    steps:
      - name: Deploy to GitHub Pages
